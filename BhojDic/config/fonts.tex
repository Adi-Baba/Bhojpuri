\usepackage{fontspec}

% Set up font fallbacks with HarfBuzz rendering engine (mode=harf) for LuaLaTeX
% so Devanagari ligatures (क्ष, ज्ञ, र्व, क्र, त्र, आदि) and Kaithi script
% render with 100% accurate OpenType script shaping.
\directlua{
  luaotfload.add_fallback(
    "bhojfallback",
    {
      "Noto Sans Devanagari:mode=harf;script=dev2;language=dflt;",
      "Noto Sans Kaithi:mode=harf;script=kthi;language=dflt;"
    }
  )
}

% Main font (Noto Sans with HarfBuzz Devanagari and Kaithi fallback)
\setmainfont{Noto Sans}[RawFeature={fallback=bhojfallback}]

% Explicit font families for direct script styling
\newfontfamily\kaithifont{Noto Sans Kaithi}[
  Renderer=HarfBuzz,
  Script=Kaithi,
  BoldFont={Noto Sans Kaithi},
  ItalicFont={Noto Sans Kaithi},
  BoldFeatures={FakeBold=1.5},
  ItalicFeatures={FakeSlant=0.2}
]
\newfontfamily\devafont{Noto Sans Devanagari}[Renderer=HarfBuzz, Script=Devanagari, Language=Hindi]

% Wrappers
\newcommand{\K}[1]{{\kaithifont #1}}
\newcommand{\D}[1]{{\devafont #1}}
